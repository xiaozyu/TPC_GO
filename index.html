function parsePowerCoord(codeStr) { let cleaned = codeStr.replace(/\s+/g, '').toUpperCase(); if (/^\d{4}[A-Z]{2}(\d{2}|\d{4})$/.test(cleaned)) { cleaned = 'G' + cleaned; } // 格式：分區(1) + 圖幅X(2) + 圖幅Y(2) + 細分X(1) + 細分Y(1) + 點位(2或4) const match = cleaned.match(/^([A-R])(\d{2})(\d{2})([A-H|J-Z])([A-E])(\d{2}|\d{4})$/); if (!match) return null; const [_, zone, xStr, yStr, subXStr, subYStr, ptStr] = match; // 台電分區原點基準 (TWD67/97 網格基準) // G區基準原點：X=170000, Y=2650000 (對齊 800m x 500m 網格) const zoneOrigins = { 'A': [250000, 2780000], 'B': [330000, 2780000], 'C': [170000, 2730000], 'D': [250000, 2730000], 'E': [90000, 2680000], 'F': [170000, 2680000], 'G': [170000, 2630000], 'H': [250000, 2630000] }; const origin = zoneOrigins[zone] || [170000, 2630000]; const xMap = parseInt(xStr, 10); const yMap = parseInt(yStr, 10); // 橫向 A~H (扣除 I)，若延伸代號則依序位移 let subX = subXStr.charCodeAt(0) - 65; if (subXStr.charCodeAt(0) > 73) subX -= 1; // 跳過 'I' // 縱向 A~E (由北向南: A=400m, B=300m, C=200m, D=100m, E=0m) const subY = 4 - (subYStr.charCodeAt(0) - 65); let twdX = origin[0] + xMap * 800 + subX * 100; let twdY = origin[1] + yMap * 500 + subY * 100; if (ptStr.length === 2) { twdX += parseInt(ptStr[0], 10) * 10; twdY += parseInt(ptStr[1], 10) * 10; } else if (ptStr.length === 4) { twdX += parseInt(ptStr.substring(0, 2), 10); twdY += parseInt(ptStr.substring(2, 4), 10); } // TWD67 基準微調至 TWD97 基準 (台電原始底圖為 TWD67，平移差約 dx: 828m, dy: -206m) const twd97X = twdX - 828; const twd97Y = twdY + 206; return { ...twd97ToWGS84(twd97X, twd97Y), standardCode: cleaned }; }
